import pygame import random import math import sys # Initialize Pygame pygame.init() # Screen Setup (Portrait Mode for Mobile) WIDTH, HEIGHT = 450, 800 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Claudia's 17th Birthday!") # --- AESTHETIC COLOR PALETTE --- BG_COLOR = (11, 7, 25) # Deep Midnight Indigo TEXT_CYAN = (0, 242, 254) # Cyber Neon Cyan TEXT_GOLD = (255, 215, 0) # Soft Glowing Gold TEXT_WHITE = (240, 240, 255) # Crisp Off-White TEXT_PINK = (255, 182, 193) # Soft Pink # Cake & Gift Palette CAKE_TOP = (255, 105, 180) # Strawberry Icing CAKE_BASE = (245, 230, 210) # Vanilla Sponge CAKE_PLATE = (100, 80, 140) # Lavender Shadow GIFT_BOX = (255, 99, 71) # Coral Red SKIN_TONE = (255, 220, 190) # Girl Skin Color HAIR_COLOR = (90, 45, 20) # Cute Brown Hair # Accents & Buttons BTN_BLOW = (138, 43, 226) # Neon Purple BTN_CUT = (255, 99, 71) # Coral Pink BTN_GIFT = (0, 206, 201) # Mint Aqua BTN_WISH = (255, 215, 0) # Gold Wish # Fonts font_title = pygame.font.SysFont("sans", 28, bold=True) font_sub = pygame.font.SysFont("sans", 16) font_btn = pygame.font.SysFont("sans", 14, bold=True) font_arrow = pygame.font.SysFont("sans", 16, bold=True) font_wish_hdr = pygame.font.SysFont("sans", 20, bold=True) font_wish_txt = pygame.font.SysFont("sans", 15, bold=True) # Falling Ambient Confetti Setup confetti = [] for _ in range(90): confetti.append({ 'x': random.randint(0, WIDTH), 'y': random.randint(-HEIGHT, 0), 'color': random.choice([(255, 105, 180), (0, 242, 254), (255, 215, 0), (138, 43, 226), (0, 206, 201)]), 'speed': random.uniform(2, 5), 'size': random.randint(3, 7) }) # Explosive Gift Burst Confetti burst_particles = [] # Balloons setup balloons = [] for _ in range(8): balloons.append({ 'x': random.randint(30, WIDTH - 30), 'y': random.randint(HEIGHT, HEIGHT + 300), 'color': random.choice([(255, 75, 145), (121, 40, 202), (0, 223, 216), (255, 215, 0)]), 'speed': random.uniform(1.5, 3.0), 'phase': random.uniform(0, 6.28) }) # Interactive States candles_blown = False cake_cut = False gift_opened = False wish_revealed = False # Buttons btn_candle = pygame.Rect(15, 730, 95, 45) btn_cake = pygame.Rect(120, 730, 95, 45) btn_gift = pygame.Rect(225, 730, 95, 45) btn_wish = pygame.Rect(330, 730, 105, 45) clock = pygame.time.Clock() frame_count = 0 def trigger_confetti_burst(): """Spawns an explosive burst of confetti from the gift box center""" for _ in range(120): angle = random.uniform(0, 6.28) speed = random.uniform(3, 12) burst_particles.append({ 'x': WIDTH // 2, 'y': 450, 'vx': math.cos(angle) * speed, 'vy': math.sin(angle) * speed - 3, 'color': random.choice([(255, 105, 180), (0, 242, 254), (255, 215, 0), (255, 255, 255), (138, 43, 226)]), 'size': random.randint(4, 9), 'life': random.randint(40, 80) }) running = True while running: clock.tick(60) frame_count += 1 screen.fill(BG_COLOR) # Touch / Event Input for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.MOUSEBUTTONDOWN: pos = event.pos if btn_candle.collidepoint(pos): candles_blown = True elif btn_cake.collidepoint(pos): cake_cut = True elif btn_gift.collidepoint(pos): if not gift_opened: gift_opened = True trigger_confetti_burst() elif btn_wish.collidepoint(pos): wish_revealed = True # 1. Floating Balloons for b in balloons: b['y'] -= b['speed'] sway = math.sin(frame_count * 0.04 + b['phase']) * 16 if b['y'] < -60: b['y'] = HEIGHT + 60 b['x'] = random.randint(30, WIDTH - 30) pygame.draw.line(screen, (180, 180, 210), (b['x'] + sway, b['y'] + 20), (b['x'] + sway, b['y'] + 60), 1) pygame.draw.ellipse(screen, b['color'], (b['x'] + sway - 20, b['y'] - 25, 40, 50)) # 2. Header Titles t1 = font_title.render("Happy 17th Birthday!", True, TEXT_CYAN) t2 = font_title.render("✨ CLAUDIA ✨", True, TEXT_GOLD) screen.blit(t1, (WIDTH//2 - t1.get_width()//2, 35)) screen.blit(t2, (WIDTH//2 - t2.get_width()//2, 75)) # 3. Cake Plate & Tiers cx, cy = WIDTH // 2, 280 pygame.draw.ellipse(screen, CAKE_PLATE, (cx - 100, cy + 50, 200, 22)) pygame.draw.rect(screen, CAKE_BASE, (cx - 80, cy, 160, 50), border_radius=10) pygame.draw.rect(screen, CAKE_TOP, (cx - 80, cy, 160, 16), border_top_left_radius=10, border_top_right_radius=10) pygame.draw.rect(screen, CAKE_BASE, (cx - 50, cy - 40, 100, 40), border_radius=10) pygame.draw.rect(screen, CAKE_TOP, (cx - 50, cy - 40, 100, 12), border_top_left_radius=10, border_top_right_radius=10) if cake_cut: pygame.draw.line(screen, BG_COLOR, (cx, cy - 40), (cx, cy + 50), 4) # 4. Candles for x in [cx - 18, cx + 18]: pygame.draw.rect(screen, TEXT_CYAN, (x - 3, cy - 60, 6, 20), border_radius=2) if not candles_blown: r = random.randint(4, 7) pygame.draw.circle(screen, TEXT_GOLD, (x, cy - 66), r) pygame.draw.circle(screen, (255, 80, 0), (x, cy - 66), max(2, r - 3)) # 5. Gift Box, Peeking Girl (Shrujal) & Arrow Indicator gx, gy = WIDTH // 2, 460 if gift_opened: head_y = gy - 38 # Hair & Face pygame.draw.circle(screen, HAIR_COLOR, (gx, head_y - 2), 24) pygame.draw.circle(screen, SKIN_TONE, (gx, head_y + 2), 18) # Eyes & Blush pygame.draw.circle(screen, (20, 20, 20), (gx - 6, head_y), 3) pygame.draw.circle(screen, (20, 20, 20), (gx + 6, head_y), 3) pygame.draw.circle(screen, (255, 150, 150), (gx - 10, head_y + 6), 3) pygame.draw.circle(screen, (255, 150, 150), (gx + 10, head_y + 6), 3) # Smile pygame.draw.arc(screen, (150, 50, 50), (gx - 5, head_y + 3, 10, 7), 3.14, 6.28, 2) # Arms raised pygame.draw.line(screen, SKIN_TONE, (gx - 16, head_y + 10), (gx - 28, head_y - 8), 4) pygame.draw.line(screen, SKIN_TONE, (gx + 16, head_y + 10), (gx + 28, head_y - 8), 4) # Gift Body & Lid pygame.draw.rect(screen, GIFT_BOX, (gx - 35, gy, 70, 50), border_radius=6) pygame.draw.rect(screen, TEXT_GOLD, (gx - 5, gy, 10, 50)) pygame.draw.rect(screen, TEXT_GOLD, (gx - 40, gy - 58, 80, 10), border_radius=4) # --- ARROW POINTING TO SHRUJAL --- arrow_start_x = gx - 105 arrow_end_x = gx - 30 arrow_y = head_y + 2 # Arrow Line pygame.draw.line(screen, TEXT_GOLD, (arrow_start_x, arrow_y), (arrow_end_x, arrow_y), 3) # Arrow Head pygame.draw.polygon(screen, TEXT_GOLD, [ (arrow_end_x, arrow_y), (arrow_end_x - 10, arrow_y - 6), (arrow_end_x - 10, arrow_y + 6) ]) # Arrow Label Text txt_shrujal = font_arrow.render("Shrujal! 👈", True, TEXT_GOLD) screen.blit(txt_shrujal, (arrow_start_x - 10, arrow_y - 22)) else: # Closed Gift Box pygame.draw.rect(screen, GIFT_BOX, (gx - 35, gy, 70, 50), border_radius=6) pygame.draw.rect(screen, TEXT_GOLD, (gx - 40, gy - 8, 80, 10), border_radius=4) pygame.draw.rect(screen, TEXT_GOLD, (gx - 5, gy, 10, 50)) # 6. BIGGER WISHES AND BLESSINGS CARD if wish_revealed: wish_card = pygame.Rect(20, 535, 410, 175) pygame.draw.rect(screen, (25, 12, 45), wish_card, border_radius=18) pygame.draw.rect(screen, TEXT_GOLD, wish_card, width=2, border_radius=18) hdr = font_wish_hdr.render("✨ Birthday Blessings & Wishes ✨", True, TEXT_GOLD) line1 = font_wish_txt.render("🌸 May your 17th year be filled with endless joy,", True, TEXT_WHITE) line2 = font_wish_txt.render("unforgettable moments, and boundless success!", True, TEXT_WHITE) line3 = font_wish_txt.render("🙏 May God bless you with health and happiness,", True, TEXT_PINK) line4 = font_wish_txt.render("and may all your dreams come true! Keep shining!", True, TEXT_PINK) screen.blit(hdr, (WIDTH//2 - hdr.get_width()//2, 548)) screen.blit(line1, (WIDTH//2 - line1.get_width()//2, 582)) screen.blit(line2, (WIDTH//2 - line2.get_width()//2, 608)) screen.blit(line3, (WIDTH//2 - line3.get_width()//2, 638)) screen.blit(line4, (WIDTH//2 - line4.get_width()//2, 664)) else: prompt_txt = font_sub.render("Tap '✨ Blessings' below to read special birthday wishes!", True, TEXT_WHITE) screen.blit(prompt_txt, (WIDTH//2 - prompt_txt.get_width()//2, 610)) # 7. Update & Draw Burst Confetti Particles for p in burst_particles[:]: p['x'] += p['vx'] p['y'] += p['vy'] p['vy'] += 0.25 # Gravity effect p['life'] -= 1 if p['life'] <= 0: burst_particles.remove(p) else: pygame.draw.circle(screen, p['color'], (int(p['x']), int(p['y'])), p['size']) # 8. Falling Ambient Confetti for c in confetti: c['y'] += c['speed'] if c['y'] > HEIGHT: c['y'] = -10 c['x'] = random.randint(0, WIDTH) pygame.draw.circle(screen, c['color'], (int(c['x']), int(c['y'])), c['size']) # 9. Touch Buttons pygame.draw.rect(screen, BTN_BLOW, btn_candle, border_radius=10) pygame.draw.rect(screen, BTN_CUT, btn_cake, border_radius=10) pygame.draw.rect(screen, BTN_GIFT, btn_gift, border_radius=10) pygame.draw.rect(screen, BTN_WISH, btn_wish, border_radius=10) b1_txt = font_btn.render("💨 Blow", True, TEXT_WHITE) b2_txt = font_btn.render("🔪 Cut", True, TEXT_WHITE) b3_txt = font_btn.render("🎁 Open", True, (10, 10, 20)) b4_txt = font_btn.render("✨ Blessings", True, (10, 10, 20)) screen.blit(b1_txt, (btn_candle.x + 18, btn_candle.y + 14)) screen.blit(b2_txt, (btn_cake.x + 22, btn_cake.y + 14)) screen.blit(b3_txt, (btn_gift.x + 18, btn_gift.y + 14)) screen.blit(b4_txt, (btn_wish.x + 12, btn_wish.y + 14)) pygame.display.flip() pygame.quit() sys.exit()
